Terminal / Script 
# Lecture native DuckDB sans copie disque ni serveur distant
import duckdb

con = duckdb.connect()
df = con.execute("""
    SELECT date, temp_mean_c, active_traffic_events
    FROM read_parquet('data/gold/gold_luxembourg_daily_context.parquet')
    ORDER BY date DESC
    LIMIT 10;
""").df()
print(df)
